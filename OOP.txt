(1)// Function Prototyping for the following functions: add(), sub(), mul(), div()
========================================================================================================
#include <iostream>
using namespace std;

double add(double a, double b);
double sub(double a, double b);
double mul(double a, double b);
double div(double a, double b);

int main(){
    double num1, num2;
    cout<<"Enter two number";
    cin>>num1>>num2;
    cout<<add(num1, num2)<<endl;
    cout<<sub(num1, num2)<<endl;
    cout<<mul(num1, num2)<<endl;
    if (num2 != 0){
        cout<<div(num1, num2)<<endl;
    }else{
        cout<<"Number can't be divide by zero";
    }

}

double add(double a, double b){
    return a + b;
}
double sub(double a, double b){
    return a - b;
}
double mul(double a, double b){
    return a * b;
}
double div(double a, double b){
    return a / b;
}

(7)// Write a CPP program to swap variables using Function Call by Reference
========================================================================================================
#include <iostream>
#include <string>
using namespace std;

void swap(int &a, int &b){
    int temp = a;
    a = b;
    b = temp;
}

int main(){
    int x = 10;
    int y = 20;
    cout<<"Before: "<<x <<" "<<y<<endl;
    swap(x, y);
    cout<<"After:  "<<x<<" "<< y<<endl;
    return 0;
}
