(5)Assignment 11
Write a C++ program using Function Prototyping for the following functions: add(), sub(), mul(), div().
========================================================================================================
#include <iostream>
using namespace std;

// Function Prototypes (Declarations)
double add(double a, double b);
double subtract(double a, double b);
double multiply(double a, double b);
double divide(double a, double b);
double square(double a);

int main() {
    double num1, num2;
    cout << "Enter the first number: ";
    cin >> num1;
    cout << "Enter the second number: ";
    cin >> num2;
    cout << "Addition: " << add(num1, num2) << endl;
    cout << "Subtraction: " << subtract(num1, num2) << endl;
    cout << "Multiplication: " << multiply(num1, num2) << endl;
    
    if (num2 != 0) {
        cout << "Division: " << divide(num1, num2) << endl;
    } else {
        cout << "Division: Cannot divide by zero." << endl;
    }

    cout << "Square of " << num1 << ": " << square(num1) << endl;
    cout << "Square of " << num2 << ": " << square(num2) << endl;

    return 0;
}

// Function Definitions
double add(double a, double b) {
    return a + b;
}

double subtract(double a, double b) {
    return a - b;
}

double multiply(double a, double b) {
    return a * b;
}

double divide(double a, double b) {
    return a / b;
}

double square(double a) {
    return a * a;
}

(12)Assignment 12
Write a CPP program to swap variables using Function Call by Reference
========================================================================================================
#include <iostream>
#include <string>
using namespace std;

void swap(int &a, int &b){
    int temp = a;
    a = b;
    b = temp;
}

int main(){
    int x = 10;
    int y = 20;
    cout<<"Before: "<<x <<" "<<y<<endl;
    swap(x, y);
    cout<<"After:  "<<x<<" "<< y<<endl;
    return 0;
}

Write a CPP program using Return by Reference.
========================================================================================================
#include <iostream>
#include <string>
using namespace std;

int& swap(int &a, int &b){
    int temp = a;
    a = b;
    b = temp;
    return a;
}

int main(){
    int x = 10;
    int y = 20;
    cout<<"Before: "<<x <<" "<<y<<endl;
    cout<<"return value: "<<swap(x, y)<<endl;
    swap(x, y);
    cout<<"After:  "<<x<<" "<< y<<endl;
    return 0;
}

(15)Assignment 12
Write a CPP program to overload add() function.
========================================================================================================
#include <iostream>
using namespace std;

// 1. Function with two integer parameters
int add(int a, int b) {
    return a + b;
}

// 2. Overloaded function with three integer parameters
int add(int a, int b, int c) {
    return a + b + c;
}

// 3. Overloaded function with two double parameters
double add(double a, double b) {
    return a + b;
}

int main() {
    cout << "Sum of 2 ints: " << add(5, 10) << endl;          // Calls version 1
    cout << "Sum of 3 ints: " << add(5, 10, 15) << endl;      // Calls version 2
    cout << "Sum of 2 doubles: " << add(2.5, 3.5) << endl;    // Calls version 3
    
    return 0;
}

Write a CPP program using Inline Function.
========================================================================================================
#include <iostream>
using namespace std;

// Inline function to calculate the square of a number
inline int square(int x) {
    return x * x;
}

int main() {
    int num1 = 5;
    int num2 = 8;

    // The compiler may replace these calls directly with the function body
    cout << "Square of " << num1 << ": " << square(num1) << endl;
    return 0;
}
