#include<iostream>
#include<string>
using namespace std;

class Data{
    private:
        char name;
        int id;
        double weight;
    public:
    Data(){
        name = 'M';
        id = 10;
        weight = 20.5;
        }

    char getName() {
        return name;
    }
    int getId() {
        return id;
    }

    double getWeight() {
        return weight;
    }
};

int main(){
    Data d;
    cout<<d.getName()<<endl;
    cout<<d.getId()<<endl;
    cout<<d.getWeight();
    return 0;
}
